Script SQL para Supabase 

CREATE TABLE "CARRERA" (
    "ID_Carrera" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre_Carrera" VARCHAR(100) NOT NULL
);

CREATE TABLE "ALUMNO" (
    "ID_Alumno" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre" VARCHAR(100) NOT NULL,
    "DNI" VARCHAR(15) UNIQUE NOT NULL,
    "ID_Carrera" BIGINT NOT NULL REFERENCES "CARRERA"("ID_Carrera") ON DELETE CASCADE
);

CREATE TABLE "DOCENTE" (
    "ID_Docente" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre_Docente" VARCHAR(100) NOT NULL
);

CREATE TABLE "CURSO" (
    "ID_Curso" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre_Curso" VARCHAR(100) NOT NULL,
    "ID_Docente" BIGINT NOT NULL REFERENCES "DOCENTE"("ID_Docente") ON DELETE CASCADE
);

CREATE TABLE "MATRICULA" (
    "ID_Alumno" BIGINT NOT NULL REFERENCES "ALUMNO"("ID_Alumno") ON DELETE CASCADE,
    "ID_Curso" BIGINT NOT NULL REFERENCES "CURSO"("ID_Curso") ON DELETE CASCADE,
    "Nota" NUMERIC(4,2),
    PRIMARY KEY ("ID_Alumno", "ID_Curso")
);
